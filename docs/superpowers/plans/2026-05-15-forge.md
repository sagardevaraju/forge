> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a working FORGE — a catastrophe decision intelligence platform — deployed to Vercel by end of day 7, evaluated against held-out 2025 events, demonstrable live to a class panel.

**Architecture:** Single Next.js 16 monorepo with mixed Edge / Node / Python runtimes on Vercel. Live data feeds → shared Monte Carlo scenario store (DuckDB) → three optimization layers (Portfolio MIP, Operational VRP, Claims Pre-Flagger) coupled by the scenario set → decision reconciler → three UI views. Heavy ML artifacts (CV features, XGBoost models) pre-computed offline on a GPU notebook and uploaded to Vercel Blob; everything else solves on-demand inside Vercel serverless under 10 seconds.

**Tech Stack:** Next.js 16 (App Router) · React 19 · TypeScript · Tailwind · Mapbox GL JS · Turso (libSQL) · Vercel Blob · Python 3.12 · PuLP+CBC · OR-Tools · XGBoost 1.7+ · PyTorch · timm · Prithvi-100M or Clay Foundation Model · OpenRouter (primary LLM) · GitHub Models PAT (fallback LLM) · DuckDB · Optuna · Pytest · Vitest · Playwright.

- [ ] **Step 1: Write `ml/xgb/hazus_curves.py`** — depth-damage curves from FEMA HAZUS technical manual (residential, single-family).

```python
"""Simplified HAZUS depth-damage curves for residential property.
Source: FEMA HAZUS-MH Hurricane Technical Manual, Table 6.1.
"""
import numpy as np

def wind_damage_ratio(peak_wind_mph: float, build_type: str) -> float:
    """Returns fraction of TIV lost given peak wind speed."""
    table = {  # (mph, ratio) breakpoints by build type
        "wood_frame":    [(80, 0.00), (110, 0.05), (130, 0.20), (155, 0.55), (180, 0.85)],
        "masonry":       [(80, 0.00), (110, 0.02), (130, 0.10), (155, 0.35), (180, 0.65)],
        "manufactured":  [(80, 0.00), (95,  0.15), (110, 0.45), (130, 0.80), (155, 0.95)],
    }
    curve = table.get(build_type, table["wood_frame"])
    return float(np.interp(peak_wind_mph, [p[0] for p in curve], [p[1] for p in curve]))

def surge_damage_ratio(surge_depth_m: float, elevation_m: float) -> float:
    above_floor = max(surge_depth_m - elevation_m, 0)
    breakpoints = [(0, 0.0), (0.3, 0.10), (1.0, 0.35), (2.0, 0.65), (4.0, 0.95)]
    return float(np.interp(above_floor, [b[0] for b in breakpoints], [b[1] for b in breakpoints]))
```

```typescript
// lib/llm/types.ts
export interface ChatRequest {
  messages: { role: 'system' | 'user' | 'assistant' | 'tool'; content: string; tool_call_id?: string; name?: string }[];
  tools?: { type: 'function'; function: { name: string; description: string; parameters: object } }[];
  stream?: boolean;
}
export interface ChatResponse {
  content: string;
  tool_calls?: { id: string; name: string; arguments: Record<string, unknown> }[];
}
export type ProviderFn = (req: ChatRequest) => Promise<ChatResponse>;
```

```typescript
// lib/llm/cascading-client.ts
import type { ChatRequest, ChatResponse, ProviderFn } from './types';

const RETRY_STATUSES = new Set([429, 500, 502, 503, 504]);

export class CascadingLLM {
  constructor(private opts: {
    primary: ProviderFn;
    fallback: ProviderFn;
    maxRetries: number;
    baseDelayMs: number;
  }) {}

  async chat(req: ChatRequest): Promise<ChatResponse> {
    try {
      return await this.withRetry(this.opts.primary, req);
    } catch (e: any) {
      if (e?.status && !RETRY_STATUSES.has(e.status)) throw e;
      return await this.withRetry(this.opts.fallback, req);
    }
  }

  private async withRetry(fn: ProviderFn, req: ChatRequest): Promise<ChatResponse> {
    let lastErr: any;
    for (let i = 0; i < this.opts.maxRetries; i++) {
      try { return await fn(req); }
      catch (e: any) {
        lastErr = e;
        if (!e?.status || !RETRY_STATUSES.has(e.status)) throw e;
        await new Promise(r => setTimeout(r, this.opts.baseDelayMs * Math.pow(2, i)));
      }
    }
    throw lastErr;
  }
}

export function makeOpenRouterProvider(apiKey: string, model: string): ProviderFn {
  return async (req) => {
    const r = await fetch('https://openrouter.ai/api/v1/chat/completions', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${apiKey}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ model, ...req }),
    });
    if (!r.ok) throw { status: r.status, body: await r.text() };
    const data = await r.json();
    return {
      content: data.choices[0].message.content ?? '',
      tool_calls: data.choices[0].message.tool_calls?.map((tc: any) => ({
        id: tc.id, name: tc.function.name, arguments: JSON.parse(tc.function.arguments),
      })),
    };
  };
}
```

```python
        "subsample": trial.suggest_float("subsample", 0.6, 1.0),
    }
    tscv = TimeSeriesSplit(n_splits=5)
    scores = []
    for tr, va in tscv.split(X):
        m = xgb.XGBRegressor(**params)
        m.fit(X[tr], y[tr])
        scores.append(pinball_loss(y[va], m.predict(X[va]), quantile))
    return -np.mean(scores)

def train_all_quantiles(X, y):
    models = {}
    for q in [0.10, 0.50, 0.90]:
        study = optuna.create_study(direction="maximize")
        study.optimize(lambda t: objective(t, X, y, q), n_trials=100)
        models[q] = xgb.XGBRegressor(**study.best_params).fit(X, y)
    return models
```

```typescript
export const runtime = 'edge';
import { CascadingLLM, makeOpenRouterProvider, makeGitHubModelsProvider } from '@/lib/llm/cascading-client';
import { TOOLS, TOOL_MAP } from '@/lib/llm/tool-registry';

export async function POST(req: Request) {
  const { messages } = await req.json();
  const llm = new CascadingLLM({
    primary: makeOpenRouterProvider(process.env.OPENROUTER_API_KEY!, process.env.LLM_PRIMARY_MODEL!),
    fallback: makeGitHubModelsProvider(process.env.GITHUB_MODELS_PAT!, process.env.LLM_FALLBACK_MODEL!),
    maxRetries: Number(process.env.LLM_RETRY_MAX ?? 3),
    baseDelayMs: Number(process.env.LLM_RETRY_BASE_MS ?? 500),
  });

  const tools = TOOLS.map(t => ({ type: 'function' as const, function: { name: t.name, description: t.description, parameters: t.parameters } }));

  let convo = [...messages];
  for (let i = 0; i < 5; i++) {
    const resp = await llm.chat({ messages: convo, tools });
    if (!resp.tool_calls?.length) {
      return new Response(resp.content, { headers: { 'Content-Type': 'text/plain' } });
    }
    convo.push({ role: 'assistant', content: resp.content, tool_calls: resp.tool_calls } as any);
    for (const tc of resp.tool_calls) {
      const result = await TOOL_MAP[tc.name].handler(tc.arguments as any);
      convo.push({ role: 'tool', tool_call_id: tc.id, name: tc.name, content: JSON.stringify(result) });
    }
  }
  return new Response('Tool-call loop exceeded', { status: 500 });
}
```

```python
import pulp

def solve(cohorts, capital_budget, max_nonrenew_pct):
    prob = pulp.LpProblem("FORGE_Portfolio", pulp.LpMaximize)
    ACTIONS = ["retain", "reprice_up", "reprice_down", "non_renew", "cede_qs", "cede_xs"]
    x = pulp.LpVariable.dicts("x", [(c["id"], a) for c in cohorts for a in ACTIONS], lowBound=0, upBound=1)

    # Sum-to-one per cohort
    for c in cohorts:
        prob += pulp.lpSum(x[(c["id"], a)] for a in ACTIONS) == 1

    # Objective: E[premium - loss - cession_cost]
    REPRICE_FACTOR = {"reprice_up": 1.15, "reprice_down": 0.90, "retain": 1.0, "non_renew": 0.0, "cede_qs": 0.5, "cede_xs": 1.0}
    CESSION_COST = {"cede_qs": 0.6, "cede_xs": 0.15}  # fraction of expected loss
    obj = []
    for c in cohorts:
        for a in ACTIONS:
            prem_eff = c["premium"] * REPRICE_FACTOR[a]
            loss_eff = c["loss_p50"] * (0 if a == "non_renew" else (0.5 if a == "cede_qs" else 1.0))
            cession  = c["loss_p50"] * CESSION_COST.get(a, 0)
            obj.append(x[(c["id"], a)] * (prem_eff - loss_eff - cession))
    prob += pulp.lpSum(obj)

    # Capital constraint: scenario-weighted VaR_99 ≤ capital_budget
    prob += pulp.lpSum(x[(c["id"], a)] * c["loss_p99"] * (0 if a == "non_renew" else 1) for c in cohorts for a in ACTIONS) <= capital_budget

    # Non-renew cap
    total_tiv = sum(c["tiv"] for c in cohorts)
    prob += pulp.lpSum(x[(c["id"], "non_renew")] * c["tiv"] for c in cohorts) <= max_nonrenew_pct * total_tiv

    prob.solve(pulp.PULP_CBC_CMD(msg=False, timeLimit=30))
    return {"status": pulp.LpStatus[prob.status], "objective": pulp.value(prob.objective), "actions": [
        {"cohort_id": c["id"], **{a: pulp.value(x[(c["id"], a)]) for a in ACTIONS}} for c in cohorts
    ]}
```

```python
import pulp

def solve_assignment(adjusters, zones, expected_claim_volume_per_zone):
    prob = pulp.LpProblem("FORGE_VRP", pulp.LpMinimize)
    y = pulp.LpVariable.dicts("y", [(a["id"], z["id"]) for a in adjusters for z in zones], cat="Binary")
    # Each adjuster -> exactly one zone
    for a in adjusters:
        prob += pulp.lpSum(y[(a["id"], z["id"])] for z in zones) == 1
    # Zone capacity
    for z in zones:
        prob += pulp.lpSum(y[(a["id"], z["id"])] for a in adjusters) <= z["capacity"]
    # Skill constraint
    for a in adjusters:
        for z in zones:
            if z.get("required_skill") and z["required_skill"] not in a["skills"]:
                prob += y[(a["id"], z["id"])] == 0
    # Drive-time constraint
    for a in adjusters:
        for z in zones:
            if drive_time(a, z) > 8 * 3600:
                prob += y[(a["id"], z["id"])] == 0
    # Objective: minimize expected response time = drive_time * claim_volume / adjusters_in_zone
    prob += pulp.lpSum(y[(a["id"], z["id"])] * drive_time(a, z) * expected_claim_volume_per_zone[z["id"]] for a in adjusters for z in zones)
    prob.solve(pulp.PULP_CBC_CMD(msg=False))
    return [{"adjuster_id": a["id"], "zone_id": next(z["id"] for z in zones if pulp.value(y[(a["id"], z["id"])]) > 0.5)} for a in adjusters]
```
